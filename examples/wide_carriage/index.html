const plot = new GysinPlot({ seed: 1961 });
const size = width * 0.022;

plot.underwood("ONE LETTER\nAT A TIME\nTHE PAGE FILLS", width * 0.09, height * 0.26, {
  size,
  pitch: size * 2.4,        // default is size * 1.2
  lineHeight: size * 5.6
});

plot.draw();
